/**
 * Get Alchemy Settings for EVM chains
 */
const getAlchemySettings = (network) => {
  const alchemyNetwork = alchemyNetworkMap[network?.id];
  if (!alchemyNetwork) {
    throw new Error(`Unsupported network chainId: ${network?.id}`);
  }
  return {
    apiKey: ALCHEMY_API_KEY,
    network: alchemyNetwork,
  };
};

/**
 * Create Alchemy Instance for EVM
 */
export const createAlchemyInstance = (network) => {
  const settings = getAlchemySettings(network);
  return new Alchemy(settings);
};

/**
 * Fetch Token Metadata (Solana) using Alchemy SDK
 */

export const fetchSolanaTokenMetadata = async (tokenAddress, apiKey) => {
  try {
    const options = {
      method: 'POST',
      url: `https://solana-mainnet.g.alchemy.com/v2/${apiKey}`,
      headers: {
        accept: 'application/json',
        'content-type': 'application/json',
      },
      data: {
        id: 1,
        jsonrpc: '2.0',
        method: 'getAccountInfo',
        params: [tokenAddress], // Token address to get metadata
      },
    };

    const response = await axios.request(options);

    const accountData = response.data?.result?.value?.data?.parsed?.info;

    if (!accountData) {
      throw new Error(`No data found for token address: ${tokenAddress}`);
    }

    return {
      name: accountData.name || "Unknown",
      symbol: accountData.symbol || "Unknown",
      logo: ShitcoinLogo, // Default placeholder logo if none is available
      decimals: accountData.decimals || 9,
    };
  } catch (error) {
    console.error(`Error fetching metadata for Solana token ${tokenAddress}:`, error);
    return {
      name: "n/a",
      symbol: "n/a",
      logo: ShitcoinLogo,
      decimals: 9,
    };
  }
};

    }

// Helper function to call Solana RPC methods using Alchemy
export const callSolanaRPC = async (methodName, params, apiKey) => {
    try {
        const url = `https://solana-mainnet.g.alchemy.com/v2/${apiKey}`;
        const payload = {
            jsonrpc: "2.0",
            id: 1,
            method: methodName,
            params: params,
        };

        // Make the POST request using axios
        const response = await axios.post(url, payload, {
            headers: {
                'accept': 'application/json',
                'content-type': 'application/json'
            }
        });

        const result = response.data?.result;

        if (!result) {
            throw new Error(`No data returned from the Solana RPC call using method: ${methodName}`);
        }

        console.log(`${methodName} result:`, result);
        return result;
    } catch (error) {
        console.error(`Error calling Solana RPC method ${methodName}:`, error);
        throw error;
    }
};
